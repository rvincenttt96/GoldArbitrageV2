    def sell(self, amount):

        price = self.get_price("sell")

        r = self.session.post(
            self.BASE_URL + '/api/v1/exchanges/sell',
            json={
                'action': 'sell',
                'amount': int(amount * 100),
                'discount_ids': [],
                'discountIds': [],
                'priceId': price.price_id,
                'total': int(price.price * int(amount * 100) // 10)
            }
        )

        return r.json()
